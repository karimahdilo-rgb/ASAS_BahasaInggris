      Jika diisi, nilai siswa akan otomatis dikirim ke Google Sheets saat ujian selesai maupun saat diskualifikasi.

      ☁️ Panduan Integrasi Google Sheets (Code.gs)

      1. Buka Google Sheets baru ➔ klik menu Ekstensi > Apps Script.
      2. Tempel kode lengkap dari file Code.gs (atau cuplikan di bawah), lalu klik Deploy > New Deployment > Web App (Execute as: Me, Who has access: Anyone).
      3. Salin URL Web App yang muncul dan tempelkan ke kolom URL Webhook di samping.
      function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("REKAP_UTAMA") || ss.getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["No","Timestamp","Kode","Nama","Kelas","Mapel","WaktuSelesai","Durasi","PGBenar","SkorPG(15)","BSBenar","SkorBS(30)","Uraian","SkorUraian(21)","TotalSkor(66)","NilaiAkhir","Status","Pelanggaran"]);
    }
    var d = JSON.parse(e.postData.contents);
    var r = sheet.getLastRow() + 1;
    var totFormula = d.disqualified ? 0 : "=IF(ISNUMBER(N" + r + "), J" + r + "+L" + r + "+N" + r + ", J" + r + "+L" + r + ")";
    var valFormula = d.disqualified ? 0 : "=IF(Q" + r + "=\"DISKUALIFIKASI\", 0, ROUND((O" + r + "/66)*100, 1))";
    sheet.appendRow([
      r - 1, new Date(), d.id, d.name, d.className, d.subject,
      d.finishedAt, d.durationUsed, d.pgCorrect + "/" + d.pgTotalQuestions, d.pgScore,
      d.bsStatementsCorrect + "/" + d.bsStatementsTotal, d.bsScore,
      d.essayMarkedDone + "/" + d.essayTotal, "", totFormula, valFormula,
      d.disqualified ? "DISKUALIFIKASI" : "SELESAI", d.violations
    ]);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
